cmake_minimum_required(VERSION 3.16)
project(chip-audio VERSION 0.6.0 LANGUAGES C)
include(GNUInstallDirs)
add_library(chip-audio STATIC src/chip-audio.c src/chip-drums.c)
add_library(chip-audio::chip-audio ALIAS chip-audio)
set_target_properties(chip-audio PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(chip-audio PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(NOT MSVC)
    target_link_libraries(chip-audio PUBLIC m)
endif()
install(TARGETS chip-audio EXPORT chip-audio-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/chip-audio)
install(EXPORT chip-audio-targets NAMESPACE chip-audio:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chip-audio)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/chip-audio-config.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/chip-audio-config.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chip-audio)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/chip-audio-config-version.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/chip-audio-config.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/chip-audio-config-version.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/chip-audio)
option(CHIP_AUDIO_BUILD_TESTS "Build chip-audio tests" OFF)
if(CHIP_AUDIO_BUILD_TESTS)
    enable_testing()
    enable_language(CXX)
    function(chip_test name source)
        add_executable(${name} ${source})
        target_link_libraries(${name} PRIVATE chip-audio)
        if(MSVC)
            target_compile_options(${name} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${name} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${name} COMMAND ${name})
    endfunction()
    chip_test(chip-audio-test tests/audio-test.c)
    chip_test(chip-audio-cxx-test tests/cxx-link-test.cpp)
    target_compile_features(chip-audio-cxx-test PRIVATE cxx_std_17)
    chip_test(chip-audio-waveform-test tests/waveform-test.c)
    chip_test(chip-audio-echo-test tests/echo-test.c)
    chip_test(chip-audio-contract-test tests/contract-test.c)
    chip_test(chip-audio-echo-plan-test tests/echo-plan-test.c)
    chip_test(chip-audio-stems-test tests/stems-test.c)
    chip_test(chip-audio-player-test tests/player-test.c)
    chip_test(chip-audio-player-ownership-test tests/player-ownership-test.c)
    # Compare independent expressions with identical floating-point rounding.
    if(MSVC)
        target_compile_options(chip-audio-waveform-test PRIVATE /fp:strict)
    else()
        target_compile_options(chip-audio-waveform-test PRIVATE -ffp-contract=off)
    endif()
endif()
